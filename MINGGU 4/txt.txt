class NotifikasiTelegram(Notifikasi):

    def kirim_pesan(self, pesan):
        print(
            f"[Telegram] Mengirim ke {self.tujuan}: {pesan}"
        )


# Uji implementasi class turunan
email = NotifikasiEmail("user@email.com")
telegram = NotifikasiTelegram("user_tele")

email.kirim("Halo, ini notifikasi email.")
telegram.kirim("Halo, ini notifikasi Telegram.")
